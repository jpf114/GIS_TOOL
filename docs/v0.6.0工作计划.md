# GIS TOOL v0.6.0 工作计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 将 GIS TOOL 从"纯参数驱动型工具面板"升级为"GIS 工作台"，补齐结果预览、任务管理、设置系统三大核心缺口，同时完善 GUI 基础设施。

**架构：** 在现有插件化架构基础上，新增 MapCanvas 组件实现栅格/矢量预览，新增 TaskManager 实现任务队列与历史，新增 SettingsManager 实现持久化配置。GUI 布局从"导航+参数"双栏扩展为"导航+预览+参数"三栏。

**技术栈：** C++17 / Qt6 Widgets / GDAL / OpenCV / vcpkg

```cpp
    QSettings settings_;
    static constexpr int kMaxRecentFiles = 10;
};
```

`src/gui/settings_manager.cpp`：
```cpp
#include "settings_manager.h"

SettingsManager& SettingsManager::instance() {
    static SettingsManager inst;
    return inst;
}

SettingsManager::SettingsManager(QObject* parent)
    : QObject(parent),
      settings_(QStringLiteral("GISTool"), QStringLiteral("GISTool")) {}

QString SettingsManager::lastInputDirectory() const {
    return settings_.value(QStringLiteral("paths/lastInputDir")).toString();
}

void SettingsManager::setLastInputDirectory(const QString& dir) {
    settings_.setValue(QStringLiteral("paths/lastInputDir"), dir);
}

QString SettingsManager::lastOutputDirectory() const {
    return settings_.value(QStringLiteral("paths/lastOutputDir")).toString();
}

void SettingsManager::setLastOutputDirectory(const QString& dir) {
    settings_.setValue(QStringLiteral("paths/lastOutputDir"), dir);
}

QStringList SettingsManager::recentFiles() const {
    return settings_.value(QStringLiteral("recent/files")).toStringList();
}

void SettingsManager::addRecentFile(const QString& path) {
    auto files = recentFiles();
    files.removeAll(path);
    files.prepend(path);
    while (files.size() > kMaxRecentFiles)
        files.removeLast();
    settings_.setValue(QStringLiteral("recent/files"), files);
    emit recentFilesChanged();
}

void SettingsManager::clearRecentFiles() {
    settings_.remove(QStringLiteral("recent/files"));
    emit recentFilesChanged();
}

QString SettingsManager::defaultOutputSrs() const {
    return settings_.value(QStringLiteral("defaults/outputSrs")).toString();
}

void SettingsManager::setDefaultOutputSrs(const QString& srs) {
    settings_.setValue(QStringLiteral("defaults/outputSrs"), srs);
}

int SettingsManager::maxRecentFiles() const {
    return kMaxRecentFiles;
}
```

```cpp
#include "settings_dialog.h"
#include "settings_manager.h"
#include "crs_dialog.h"
#include <QFormLayout>
#include <QLineEdit>
#include <QPushButton>
#include <QDialogButtonBox>
#include <QLabel>
#include <QGroupBox>

SettingsDialog::SettingsDialog(QWidget* parent)
    : QDialog(parent) {
    setWindowTitle(QStringLiteral("设置"));
    setMinimumWidth(400);

    auto* layout = new QFormLayout(this);

    auto* srsEdit = new QLineEdit;
    srsEdit->setPlaceholderText(QStringLiteral("如 EPSG:4326"));
    srsEdit->setText(SettingsManager::instance().defaultOutputSrs());
    auto* srsBtn = new QPushButton(QStringLiteral("选择"));
    auto* srsLayout = new QHBoxLayout;
    srsLayout->addWidget(srsEdit);
    srsLayout->addWidget(srsBtn);
    connect(srsBtn, &QPushButton::clicked, this, [srsEdit]() {
        CrsDialog dlg;
        if (dlg.exec() == QDialog::Accepted) {
            QString crs = dlg.selectedCrs();
            if (!crs.isEmpty()) srsEdit->setText(crs);
        }
    });
    layout->addRow(QStringLiteral("默认输出坐标系："), srsLayout);

    auto* btnBox = new QDialogButtonBox(QDialogButtonBox::Ok | QDialogButtonBox::Cancel);
    layout->addRow(btnBox);

    connect(btnBox, &QDialogButtonBox::accepted, this, [this, srsEdit]() {
        SettingsManager::instance().setDefaultOutputSrs(srsEdit->text());
        accept();
    });
    connect(btnBox, &QDialogButtonBox::rejected, this, &QDialog::reject);
}
```

```cpp
#pragma once
#include <QObject>

#include <QDateTime>
#include <vector>
#include <map>
#include <string>
#include <gis/framework/result.h>
#include <gis/framework/param_spec.h>

struct TaskRecord {
    QString id;
    QString pluginName;
    QString actionKey;
    std::map<std::string, gis::framework::ParamValue> params;
    gis::framework::Result result;
    QDateTime startTime;
    QDateTime endTime;
    enum Status { Pending, Running, Completed, Cancelled, Failed } status = Pending;
};

class TaskManager : public QObject {
    Q_OBJECT
public:
    static TaskManager& instance();

    QString submitTask(const QString& pluginName,
                       const QString& actionKey,
                       const std::map<std::string, gis::framework::ParamValue>& params);
    const TaskRecord* findTask(const QString& id) const;
    const std::vector<TaskRecord>& history() const;
    void clearHistory();

signals:
    void taskSubmitted(const QString& id);
    void taskStarted(const QString& id);
    void taskFinished(const QString& id);

private:
    TaskManager(QObject* parent = nullptr);
    std::vector<TaskRecord> history_;
    int nextId_ = 1;
};
```
