创建 `tests/raster_fixture.h`：

```cpp
#pragma once

#include <array>
#include <filesystem>
#include <string>
#include <vector>
#include <gdal_priv.h>
#include <ogrsf_frmts.h>
#include <gis/core/gdal_wrapper.h>

namespace gis::tests {

enum class FixtureCrs { None, Epsg4326, Epsg3857 };

struct RasterFixtureOptions {
    FixtureCrs crs = FixtureCrs::None;
    bool hasNoData = false;
    double noData = -9999.0;
    GDALDataType dtype = GDT_Float32;
    double originX = 116.0;
    double originY = 40.0;
    double pixelSize = 0.001;
};

inline int epsgFor(FixtureCrs crs) {
    switch (crs) {
        case FixtureCrs::Epsg4326: return 4326;
        case FixtureCrs::Epsg3857: return 3857;
        default: return 0;
    }
}

inline void applyFixtureHeader(GDALDataset* ds, const RasterFixtureOptions& options,
                               int width, int height) {
    double gt[6] = {options.originX, options.pixelSize, 0.0,
                    options.originY, 0.0, -options.pixelSize};
    ds->SetGeoTransform(gt);
    const int epsg = epsgFor(options.crs);
    if (epsg > 0) {
        OGRSpatialReference srs;
        srs.importFromEPSG(epsg);
        char* wkt = nullptr;
        srs.exportToWkt(&wkt);
        ds->SetProjection(wkt ? wkt : "");
        CPLFree(wkt);
    }
    if (options.hasNoData) {
        for (int b = 1; b <= ds->GetRasterCount(); ++b) {
            ds->GetRasterBand(b)->SetNoDataValue(options.noData);
        }
    }
    (void)width;
    (void)height;
}

inline std::string writeFloatRaster(const std::filesystem::path& dir, const std::string& fileName,
                                    int width, int height, const std::vector<float>& data,
                                    const RasterFixtureOptions& options = {}) {
    const auto path = (dir / fileName).string();
    auto ds = gis::core::createRaster(path, width, height, 1, options.dtype);
    applyFixtureHeader(ds.get(), options, width, height);
    auto* band = ds->GetRasterBand(1);
    band->RasterIO(GF_Write, 0, 0, width, height, const_cast<float*>(data.data()),
                   width, height, options.dtype, 0, 0);
    band->FlushCache();
    return path;
}

inline std::string writeConstantBandsRaster(const std::filesystem::path& dir,
                                            const std::string& fileName,
                                            int width, int height,
                                            const std::vector<float>& bandValues,
                                            const RasterFixtureOptions& options = {}) {
    const auto path = (dir / fileName).string();
    auto ds = gis::core::createRaster(path, width, height,
                                      static_cast<int>(bandValues.size()), options.dtype);
    applyFixtureHeader(ds.get(), options, width, height);
    std::vector<float> row(static_cast<size_t>(width), 0.0f);
    for (size_t i = 0; i < bandValues.size(); ++i) {
        std::fill(row.begin(), row.end(), bandValues[i]);
        auto* band = ds->GetRasterBand(static_cast<int>(i) + 1);
        for (int y = 0; y < height; ++y) {
            band->RasterIO(GF_Write, 0, y, width, 1, row.data(), width, 1, options.dtype, 0, 0);
        }
        band->FlushCache();
    }
    return path;
}

inline float readPixel(const std::string& path, int bandIndex, int x, int y) {
    auto ds = gis::core::openRaster(path, true);
    auto* band = ds->GetRasterBand(bandIndex);
    EXPECT_NE(band, nullptr);
    float value = 0.0f;
    band->RasterIO(GF_Read, x, y, 1, 1, &value, 1, 1, GDT_Float32, 0, 0);
    return value;
}

inline int readBandCount(const std::string& path) {
    return gis::core::openRaster(path, true)->GetRasterCount();
}

inline std::string readCrsAuthorityCode(const std::string& path) {
    auto ds = gis::core::openRaster(path, true);
    auto* srs = ds->GetSpatialRef();
    if (!srs) return {};
    const char* code = srs->GetAuthorityCode(nullptr);
    if (!code) return {};
    std::string auth = srs->GetAuthorityName(nullptr) ? srs->GetAuthorityName(nullptr) : "";
    if (auth.empty()) auth = "EPSG";
    return auth + ":" + code;
}

inline std::array<double, 6> readGeoTransform(const std::string& path) {
    auto ds = gis::core::openRaster(path, true);
    std::array<double, 6> gt{0, 0, 0, 0, 0, 0};
    ds->GetGeoTransform(gt.data());
    return gt;
}

inline bool readNoDataValue(const std::string& path, int bandIndex, double& outNoData) {
    auto ds = gis::core::openRaster(path, true);
    int found = 0;
    outNoData = ds->GetRasterBand(bandIndex)->GetNoDataValue(&found);
    return found != 0;
}

} // namespace gis::tests
```

```cpp
#include <gis/core/expression_engine.h>

#include <cctype>
#include <cstdlib>
#include <stack>

namespace gis::core {
namespace {

int precedence(ExpressionNode::Op op) {
    switch (op) {
        case ExpressionNode::Op::Add:
        case ExpressionNode::Op::Subtract: return 1;
        case ExpressionNode::Op::Multiply:
        case ExpressionNode::Op::Divide: return 2;
        case ExpressionNode::Op::Negate: return 3;
        default: return 0;
    }
}

bool isVariableStart(char c) { return std::isalpha(static_cast<unsigned char>(c)) || c == '_'; }
bool isVariableChar(char c) { return std::isalnum(static_cast<unsigned char>(c)) || c == '_'; }

} // namespace

bool compileExpression(const std::string& expression,
                       const std::vector<std::string>& variableNames,
                       std::vector<ExpressionNode>& program,
                       std::string& error) {
    program.clear();
    error.clear();
    if (expression.empty()) {
        error = "表达式为空";
        return false;
    }

    std::stack<ExpressionNode::Op> ops;
    std::vector<ExpressionNode> output;
    bool expectOperand = true;
    size_t i = 0;
    int depth = 0;

    auto findVariable = [&](const std::string& name) -> int {
        for (size_t k = 0; k < variableNames.size(); ++k) {
            if (variableNames[k] == name) return static_cast<int>(k);
        }
        return -1;
    };

    auto pushOperator = [&](ExpressionNode::Op op) {
        while (!ops.empty() && ops.top() != ExpressionNode::Op::Variable) {
            if (precedence(ops.top()) >= precedence(op)) {
                ExpressionNode node;
                node.op = ops.top();
                output.push_back(node);
                ops.pop();
            } else {
                break;
            }
        }
        ops.push(op);
    };

    while (i < expression.size()) {
        const char c = expression[i];
        if (std::isspace(static_cast<unsigned char>(c))) { ++i; continue; }

        if (std::isdigit(static_cast<unsigned char>(c)) || c == '.') {
            if (expectOperand) { error = "表达式语法错误：期望操作数"; return false; }
            const size_t start = i;
            while (i < expression.size() &&
                   (std::isdigit(static_cast<unsigned char>(expression[i])) || expression[i] == '.')) {
                ++i;
            }
            const std::string text = expression.substr(start, i - start);
            char* end = nullptr;
            const double value = std::strtod(text.c_str(), &end);
            if (end == text.c_str() || *end != '\0') { error = "非法数字: " + text; return false; }
            ExpressionNode node;
            node.op = ExpressionNode::Op::Number;
            node.number = value;
            output.push_back(node);
            expectOperand = false;
            continue;
        }

        if (isVariableStart(c)) {
            const size_t start = i;
            while (i < expression.size() && isVariableChar(expression[i])) ++i;
            const std::string name = expression.substr(start, i - start);
            const int index = findVariable(name);
            if (index < 0) { error = "未知变量: " + name; return false; }
            if (expectOperand) { error = "表达式语法错误：期望操作数"; return false; }
            ExpressionNode node;
            node.op = ExpressionNode::Op::Variable;
            node.variableIndex = index;
            output.push_back(node);
            expectOperand = false;
            continue;
        }

        if (c == '(') {
            ops.push(ExpressionNode::Op::Variable);  // 作为左括号哨兵
            ++depth;
            expectOperand = true;
            ++i;
            continue;
        }

        if (c == ')') {
            if (depth <= 0) { error = "括号不配对：多余的右括号"; return false; }
            while (!ops.empty() && ops.top() != ExpressionNode::Op::Variable) {
                ExpressionNode node;
                node.op = ops.top();
                output.push_back(node);
                ops.pop();
            }
            if (ops.empty()) { error = "括号不配对"; return false; }
            ops.pop();
            --depth;
            expectOperand = false;
            ++i;
            continue;
        }

        ExpressionNode::Op op;
        bool unary = false;
        switch (c) {
            case '+': op = expectOperand ? ExpressionNode::Op::Negate : ExpressionNode::Op::Add; unary = expectOperand; break;
            case '-': op = expectOperand ? ExpressionNode::Op::Negate : ExpressionNode::Op::Subtract; unary = true; break;
            case '*': op = ExpressionNode::Op::Multiply; break;
            case '/': op = ExpressionNode::Op::Divide; break;
            default:
                error = std::string("非法字符: ") + c;
                return false;
        }
        if (!expectOperand || !unary || op != ExpressionNode::Op::Negate) {
            if (expectOperand && op != ExpressionNode::Op::Negate) {
                error = "表达式语法错误：缺少操作数";
                return false;
            }
            pushOperator(op);
            expectOperand = true;
        } else {
            pushOperator(ExpressionNode::Op::Negate);
            expectOperand = true;
        }
        ++i;
    }

    if (expectOperand) { error = "表达式结尾缺少操作数"; return false; }
    while (!ops.empty()) {
        if (ops.top() == ExpressionNode::Op::Variable) { error = "括号不配对：缺少右括号"; return false; }
        ExpressionNode node;
        node.op = ops.top();
        output.push_back(node);
        ops.pop();
    }
    if (depth != 0) { error = "括号不配对"; return false; }
    program = std::move(output);
    return true;
}

double evaluateExpression(const std::vector<ExpressionNode>& program,
                          const std::vector<double>& values,
                          bool& evaluationFailed) {
    evaluationFailed = false;
    std::vector<double> stack;
    stack.reserve(program.size());

    for (const auto& node : program) {
        switch (node.op) {
            case ExpressionNode::Op::Number: stack.push_back(node.number); break;
            case ExpressionNode::Op::Variable: {
                if (node.variableIndex < 0 ||
                    static_cast<size_t>(node.variableIndex) >= values.size()) {
                    evaluationFailed = true;
                    return 0.0;
                }
                stack.push_back(values[static_cast<size_t>(node.variableIndex)]);
                break;
            }
            case ExpressionNode::Op::Negate: {
                if (stack.empty()) { evaluationFailed = true; return 0.0; }
                stack.back() = -stack.back();
                break;
            }
            case ExpressionNode::Op::Add:
            case ExpressionNode::Op::Subtract:
            case ExpressionNode::Op::Multiply:
            case ExpressionNode::Op::Divide: {
                if (stack.size() < 2) { evaluationFailed = true; return 0.0; }
                const double b = stack.back(); stack.pop_back();
                const double a = stack.back(); stack.pop_back();
                double result = 0.0;
                switch (node.op) {
                    case ExpressionNode::Op::Add: result = a + b; break;
                    case ExpressionNode::Op::Subtract: result = a - b; break;
                    case ExpressionNode::Op::Multiply: result = a * b; break;
                    case ExpressionNode::Op::Divide:
                        if (std::abs(b) < 1e-15) { evaluationFailed = true; result = 0.0; }
                        else { result = a / b; }
                        break;
                    default: break;
                }
                stack.push_back(result);
                break;
            }
        }
    }

    if (stack.size() != 1) { evaluationFailed = true; return 0.0; }
    return stack.front();
}

} // namespace gis::core
```

```cpp
        result = elevation.clone();
        const int rows = result.rows;
        const int cols = result.cols;
        std::vector<uint8_t> inQueue(static_cast<size_t>(rows * cols), 0);
        std::vector<std::pair<float, int>> heap;   // (出水面高程, 线性索引)

        auto linear = [cols](int y, int x) { return y * cols + x; };
        auto validCell = [&](int y, int x) {
            return y >= 0 && x >= 0 && y < rows && x < cols &&
                   (!hasNoData || noDataMask.at<uint8_t>(y, x) == 0);
        };

        // 1) 数据集边缘作为出海口入队
        for (int x = 0; x < cols; ++x) {
            for (int y : {0, rows - 1}) {
                if (!validCell(y, x) || inQueue[static_cast<size_t>(linear(y, x))]) continue;
                inQueue[static_cast<size_t>(linear(y, x))] = 1;
                heap.emplace_back(result.at<float>(y, x), linear(y, x));
            }
        }
        for (int y = 0; y < rows; ++y) {
            for (int x : {0, cols - 1}) {
                if (!validCell(y, x) || inQueue[static_cast<size_t>(linear(y, x))]) continue;
                inQueue[static_cast<size_t>(linear(y, x))] = 1;
                heap.emplace_back(result.at<float>(y, x), linear(y, x));
            }
        }
        std::make_heap(heap.begin(), heap.end());

        // 2) 从低向高淹没，洼地抬到出水口高程
        static const int kDx[8] = {1, 1, 0, -1, -1, -1, 0, 1};
        static const int kDy[8] = {0, 1, 1, 1, 0, -1, -1, -1};
        int processed = 0;
        while (!heap.empty()) {
            std::pop_heap(heap.begin(), heap.end());
            const auto top = heap.back();
            heap.pop_back();
            const int cy = top.second / cols;
            const int cx = top.second % cols;

            for (int d = 0; d < 8; ++d) {
                const int ny = cy + kDy[d];
                const int nx = cx + kDx[d];
                if (!validCell(ny, nx)) continue;
                const int idx = linear(ny, nx);
                if (inQueue[static_cast<size_t>(idx)]) continue;
                inQueue[static_cast<size_t>(idx)] = 1;

                float& value = result.at<float>(ny, nx);
                if (value < top.first) value = top.first;
                else {
                    // 已高于出水面，保持原值（不降低地形）
                }
                if (epsilon > 0.0 && value > top.first) {
                    value = std::max(value, top.first + epsilon);
                }
                heap.emplace_back(value, idx);
                std::push_heap(heap.begin(), heap.end());
            }
            if ((++processed % 4096) == 0) {
                progress.throwIfCancelled();
                progress.onProgress(0.3 + 0.6 * static_cast<double>(processed) /
                                           static_cast<double>(std::max(1, rows * cols)));
            }
        }
        if (hasNoData) {
            result.setTo(static_cast<float>(noDataValue), noDataMask);
        }
        progress.onProgress(0.9);
```

`epsilon` 参数读取（放在 `runHydrologyTerrainProcess` 参数解析处）：


```cpp
#include <gis/core/tile_grid.h>
#include <algorithm>
#include <cmath>

namespace gis::core {

constexpr double kMaxWebMercator = 20037508.342789244;

TileGrid mercatorTileGrid(int zoom, int tileSize) {
    TileGrid grid;
    const int count = std::max(1, tileSize) * (1 << std::max(0, zoom));
    grid.worldMinX = -kMaxWebMercator;
    grid.worldMinY = -kMaxWebMercator;
    grid.worldMaxX = kMaxWebMercator;
    grid.worldMaxY = kMaxWebMercator;
    grid.globalWidth = count;
    grid.globalHeight = count;
    return grid;
}

TileWindow tileWindow(const TileGrid& grid, int tileSize, int tileX, int tileY) {
    TileWindow win;
    const int tiles = grid.globalWidth / std::max(1, tileSize);
    if (tileX < 0 || tileY < 0 || tileX >= tiles || tileY >= tiles) return win;
    win.xOff = tileX * tileSize;
    win.yOff = tileY * tileSize;
    win.xSize = std::min(tileSize, grid.globalWidth - win.xOff);
    win.ySize = std::min(tileSize, grid.globalHeight - win.yOff);
    win.valid = win.xSize > 0 && win.ySize > 0;
    return win;
}

} // namespace gis::core
```
